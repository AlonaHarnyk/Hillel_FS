import { sum, isAdult, getUser, throwError, logger } from "../src/functions.js";

describe("Навчальний приклад Jasmine", () => {
  let value;

  // beforeEach — виконується перед ВСІМА тестами
  beforeAll(() => {
    console.log("Start!");
    value = 10;
  });

  // afterEach — виконується ПІСЛЯ ВСІХ тестів
  afterAll(() => {
    console.log("End");
    value = null;
  });

  // beforeEach — виконується перед КОЖНИМ тестом
  beforeEach(() => {
    value = 10;
  });

  // afterEach — виконується ПІСЛЯ КОЖНОГО тесту
  afterEach(() => {
    value = null;
  });

  // it + expect + toBe
  it("sum додає два числа (toBe)", () => {
    expect(sum(2, 3)).toBe(5);
  });

  // toEqual
  it("getUser повертає обʼєкт (toEqual)", () => {
    expect(getUser()).toEqual({ name: "Anna", age: 20 });
  });

  // toBeTruthy
  it("isAdult повертає true для дорослого (toBeTruthy)", () => {
    expect(isAdult(20)).toBeTruthy();
  });

  // toBeFalsy
  it("isAdult повертає false для неповнолітнього (toBeFalsy)", () => {
    expect(isAdult(15)).toBeFalsy();
  });

  // toContain (рядок)
  it("рядок містить підрядок (toContain)", () => {
    expect("Hello world").toContain("world");
  });

  // toContain (масив)
  it("масив містить елемент (toContain)", () => {
    expect([1, 2, 3]).toContain(2);
  });

  // toThrow
  it("функція кидає помилку (toThrow)", () => {
    expect(() => throwError()).toThrow();
  });

  // beforeEach / afterEach використання
  it("value ініціалізується в beforeEach", () => {
    expect(value).toBe(10);
  });

  // spyOn
  it("spyOn відстежує виклик методу", () => {
    spyOn(logger, "log");

    logger.log("Test message");

    expect(logger.log).toHaveBeenCalled();
  });

  // jasmine.createSpy
  it("jasmine.createSpy створює порожній spy", () => {
    const mySpy = jasmine.createSpy("mySpy");

    mySpy("hello");

    expect(mySpy).toHaveBeenCalled();
  });

  // jasmine.createSpyObj
  it("jasmine.createSpyObj створює обʼєкт зі spy", () => {
    const api = jasmine.createSpyObj("api", ["get", "save"]);

    api.get();
    api.save();

    expect(api.get).toHaveBeenCalled();
    expect(api.save).toHaveBeenCalled();
  });
});
